cmake_minimum_required(VERSION 3.28)
project(am_s7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(nlohmann_json_schema_validator CONFIG REQUIRED)
find_package(RapidJSON CONFIG REQUIRED)
find_package(jsoncons CONFIG REQUIRED)
find_package(minizip-ng CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
enable_testing()
get_filename_component(S7_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(s7_core STATIC src/reference.cpp src/candidates.cpp src/jsoncons_candidate.cpp src/zip.cpp src/platform_memory.cpp)
target_include_directories(s7_core PUBLIC src PRIVATE ${RAPIDJSON_INCLUDE_DIRS})
target_link_libraries(s7_core PUBLIC nlohmann_json::nlohmann_json nlohmann_json_schema_validator::validator
  jsoncons::jsoncons MINIZIP::minizip-ng)
if(WIN32)
  target_link_libraries(s7_core PRIVATE psapi)
endif()
add_executable(s7_run src/main.cpp)
target_link_libraries(s7_run PRIVATE s7_core)
target_compile_definitions(s7_run PRIVATE S7_ROOT="${S7_ROOT}")
add_executable(s7_unit src/test_reference.cpp)
target_link_libraries(s7_unit PRIVATE s7_core GTest::gtest_main)
foreach(target s7_core s7_run s7_unit)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /external:W0)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()
add_test(NAME s7_tokenizer COMMAND s7_unit)
cmake_host_system_information(RESULT S7_CPU QUERY PROCESSOR_DESCRIPTION)
cmake_host_system_information(RESULT S7_RAM QUERY TOTAL_PHYSICAL_MEMORY)
file(WRITE "${CMAKE_BINARY_DIR}/configuration.json"
  "{\"cpu\":\"${S7_CPU}\",\"ram_mib\":${S7_RAM},\"build_type\":\"${CMAKE_BUILD_TYPE}\",\"compiler\":\"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\",\"baseline\":\"9eec002df1cd109f8f104cd966a763daf9751489\"}\n")
execute_process(COMMAND ${Python3_EXECUTABLE} -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_fixtures.py"
  --out "${CMAKE_BINARY_DIR}/fixtures" RESULT_VARIABLE fixture_result)
if(NOT fixture_result EQUAL 0)
  message(FATAL_ERROR "S7 fixture/schema check failed")
endif()
add_test(NAME s7_guard COMMAND ${Python3_EXECUTABLE} -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/run_suite.py"
  guard --runner $<TARGET_FILE:s7_run> --build "${CMAKE_BINARY_DIR}")
add_test(NAME s7_matrix COMMAND ${Python3_EXECUTABLE} -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/run_suite.py"
  matrix --runner $<TARGET_FILE:s7_run> --build "${CMAKE_BINARY_DIR}")
add_test(NAME s7_reject COMMAND ${Python3_EXECUTABLE} -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/run_suite.py"
  reject --runner $<TARGET_FILE:s7_run> --build "${CMAKE_BINARY_DIR}")
set_tests_properties(s7_reject PROPERTIES DEPENDS s7_matrix)
add_test(NAME s7_fixtures_schema COMMAND ${Python3_EXECUTABLE} -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_fixtures.py"
  --check --out "${CMAKE_BINARY_DIR}/fixtures")
set_tests_properties(s7_tokenizer s7_guard s7_matrix s7_reject s7_fixtures_schema PROPERTIES LABELS "req:ADR-0037")
